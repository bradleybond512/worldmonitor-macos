This method returns the current positions of all visible entities in a given layer, packaged for trail rendering. Add after the existing query methods:

```typescript
/**
 * Returns position+timestamp pairs for visible entities in the named layer.
 * Used by GlobeTrails to build comet tails.
 */
getEntityPositionHistory(layerName: string): { id: string; lat: number; lon: number; timeMs: number }[] {
  const layer = this.layers.get(layerName);
  if (!layer?.source.show) return [];
  const results: { id: string; lat: number; lon: number; timeMs: number }[] = [];
  const now = JulianDate.fromDate(new Date());
  for (const entity of layer.source.entities.values) {
    if (!entity?.show) continue;
    const pos = entity.position?.getValue(now);
    if (!pos) continue;
    const carto = Cartographic.fromCartesian(pos);
    let timeMs = Date.now();
    try {
      const bag = entity.properties?.getValue(now) as { timestamp?: Date | string | number } | undefined;
      const ts = bag?.timestamp;
      if (ts) timeMs = ts instanceof Date ? ts.getTime() : Number(ts);
    } catch { /* no timestamp — use now */ }
    results.push({
      id: entity.id,
      lat: CesiumMath.toDegrees(carto.latitude),
      lon: CesiumMath.toDegrees(carto.longitude),
      timeMs,
    });
  }
  return results;
}
```

- [ ] **Step 4: Add `getEventBlocks()` method**

Returns time-range blocks for the swimlane. Groups entities by layer into event windows:

```typescript
/** Swimlane event block for a category lane. */
export interface EventBlock {
  id: string;
  layerName: string;
  category: 'conflicts' | 'disasters' | 'military' | 'seismic' | 'cyber' | 'weather';
  startMs: number;
  endMs: number;
  severity: number;
  name: string;
  lat?: number;
  lon?: number;
  isForecast: boolean;
}

/** Map of swimlane category to layer names */
const SWIMLANE_CATEGORY_MAP: Record<string, string[]> = {
  conflicts: ['conflicts', 'airstrikes'],
  disasters: ['gdacs', 'cyclones'],
  military: ['flights', 'vessels', 'darkVessels'],
  seismic: ['earthquakes', 'volcanoes'],
  cyber: ['cyber', 'gpsJamming'],
  weather: ['fires'],
};

/**
 * Returns event blocks grouped by swimlane category for timeline rendering.
 */
getEventBlocks(): EventBlock[] {
  const nowMs = Date.now();
  const blocks: EventBlock[] = [];
  const now = JulianDate.fromDate(new Date());
  for (const [category, layerNames] of Object.entries(SWIMLANE_CATEGORY_MAP)) {
    for (const layerName of layerNames) {
      const layer = this.layers.get(layerName);
      if (!layer) continue;
      for (const entity of layer.source.entities.values) {
        if (!entity) continue;
        const pos = entity.position?.getValue(now);
        if (!pos) continue;
        const carto = Cartographic.fromCartesian(pos);
        let timeMs = nowMs;
        try {
          const bag = entity.properties?.getValue(now) as { timestamp?: Date | string | number } | undefined;
          const ts = bag?.timestamp;
          if (ts) timeMs = ts instanceof Date ? ts.getTime() : Number(ts);
        } catch { /* use now */ }
        blocks.push({
          id: entity.id,
          layerName,
          category: category as EventBlock['category'],
          startMs: timeMs,
          endMs: timeMs + 3_600_000, // default 1h duration; refined per layer later
          severity: ({ airstrikes: 10, conflicts: 8, cyber: 6, earthquakes: 5, gdacs: 7, cyclones: 6, fires: 4, gpsJamming: 5 } as Record<string, number>)[layerName] ?? 1,
          name: (entity.description?.getValue(now) as string) ?? entity.name ?? layerName,
          lat: CesiumMath.toDegrees(carto.latitude),
          lon: CesiumMath.toDegrees(carto.longitude),
          isForecast: timeMs > nowMs,
        });
      }
    }
  }
  return blocks;
}
```

```typescript
import type { EventBlock } from '../GlobeDataManager.js';

/** Swimlane lane configuration */
interface LaneConfig {
  key: string;
  label: string;
  color: string;
}

const LANES: LaneConfig[] = [
  { key: 'conflicts', label: 'CONFLICTS', color: '#ff3333' },
  { key: 'disasters', label: 'DISASTERS', color: '#ffa500' },
  { key: 'military', label: 'MILITARY', color: '#00c8ff' },
  { key: 'seismic', label: 'SEISMIC', color: '#ffc800' },
  { key: 'cyber', label: 'CYBER', color: '#00ffaa' },
  { key: 'weather', label: 'WEATHER', color: '#8682ff' },
];

const ZOOM_PRESETS_MS: Record<string, number> = {
  '1h': 3_600_000,
  '6h': 21_600_000,
  '24h': 86_400_000,
  '7d': 604_800_000,
  '30d': 2_592_000_000,
};

const DEFAULT_ZOOM = '24h';

export class GlobeSwimlane {
  private container: HTMLElement;
  private root: HTMLElement | null = null;
  private lanesContainer: HTMLElement | null = null;
  private nowLine: HTMLElement | null = null;
  private tooltip: HTMLElement | null = null;
  private collapsed = false;
  private zoomKey = DEFAULT_ZOOM;
  private zoomMs = ZOOM_PRESETS_MS[DEFAULT_ZOOM]!;
  private nowMs = Date.now();
  private destroyed = false;
  private blocks: EventBlock[] = [];
  private zoomPills: Map<string, HTMLElement> = new Map();
  private laneTrackEls: Map<string, HTMLElement> = new Map();
  private laneLabels: Map<string, HTMLElement> = new Map();
  private refreshTimeout: ReturnType<typeof setTimeout> | null = null;

  private _onTimeChange: ((ms: number) => void) | null = null;
  private _onEventClick: ((block: EventBlock) => void) | null = null;
  private _onLaneToggle: ((category: string) => void) | null = null;

  constructor(container: HTMLElement) {
    this.container = container;
  }

  setOnTimeChange(cb: (ms: number) => void): void { this._onTimeChange = cb; }
  setOnEventClick(cb: (block: EventBlock) => void): void { this._onEventClick = cb; }
  setOnLaneToggle(cb: (category: string) => void): void { this._onLaneToggle = cb; }

  mount(): void {
    this.root = document.createElement('div');
    this.root.className = 'ge-swimlane expanded';
    this.buildHeader();
    this.buildLanes();
    this.buildTimeAxis();
    this.buildTooltip();
    this.container.append(this.root);
    this.startNowLineUpdate();
  }

  destroy(): void {
    this.destroyed = true;
    if (this.refreshTimeout != null) { clearTimeout(this.refreshTimeout); this.refreshTimeout = null; }
    this.root?.remove();
    this.root = null;
  }

  updateBlocks(blocks: EventBlock[]): void {
    this.blocks = blocks;
    this.renderBlocks();
  }

  updateNow(ms: number): void {
    this.nowMs = ms;
  }

  toggleCollapse(): void {
    this.collapsed = !this.collapsed;
    if (!this.root) return;
    this.root.className = `ge-swimlane ${this.collapsed ? 'collapsed' : 'expanded'}`;
    if (this.lanesContainer) {
      this.lanesContainer.style.display = this.collapsed ? 'none' : '';
    }
  }

  zoomIn(): void {
    const keys = Object.keys(ZOOM_PRESETS_MS);
    const idx = keys.indexOf(this.zoomKey);
    if (idx > 0) this.setZoom(keys[idx - 1]!);
  }

  zoomOut(): void {
    const keys = Object.keys(ZOOM_PRESETS_MS);
    const idx = keys.indexOf(this.zoomKey);
    if (idx < keys.length - 1) this.setZoom(keys[idx + 1]!);
  }

  get isCollapsed(): boolean { return this.collapsed; }

  private setZoom(key: string): void {
    this.zoomKey = key;
    this.zoomMs = ZOOM_PRESETS_MS[key] ?? ZOOM_PRESETS_MS[DEFAULT_ZOOM]!;
    for (const [k, el] of this.zoomPills) {
```

## Task 5: GlobePlayback — Three Playback Engines

**Files:**
- Create: `src/components/gods-eye/GlobePlayback.ts`

Wraps time advancement logic for the three cinematic playback modes. Takes GlobeTimeMachine as its time driver and AutoFollowEngine for AI Director mode.

- [ ] **Step 1: Create GlobePlayback.ts**

```typescript
import type { GlobeTimeMachine } from '../GlobeTimeMachine.js';
import type { AutoFollowEngine } from './AutoFollowEngine.js';
import type { EventBlock } from '../GlobeDataManager.js';

export type PlaybackMode = 'documentary' | 'director' | 'heartbeat';

interface HeartbeatWindow {
  startMs: number;
  endMs: number;
  density: number; // events per hour in this window
}

export class GlobePlayback {
  private timeMachine: GlobeTimeMachine;
  private autoFollow: AutoFollowEngine | null;
  private _mode: PlaybackMode = 'documentary';
  private rafId: number | null = null;
  private lastFrame = 0;
  private _playing = false;
  private destroyed = false;
  private eventBlocks: EventBlock[] = [];
  private heartbeatWindows: HeartbeatWindow[] = [];

  // Heartbeat pacing parameters
  private static readonly HB_MIN_SPEED = 0.5;
  private static readonly HB_MAX_SPEED = 128;
  private static readonly HB_WINDOW_MS = 3_600_000; // 1h buckets

  private _onModeChange: ((mode: PlaybackMode) => void) | null = null;

  constructor(timeMachine: GlobeTimeMachine, autoFollow: AutoFollowEngine | null) {
    this.timeMachine = timeMachine;
    this.autoFollow = autoFollow;
  }

  get mode(): PlaybackMode { return this._mode; }
  get playing(): boolean { return this._playing; }

  setOnModeChange(cb: (mode: PlaybackMode) => void): void { this._onModeChange = cb; }

  setMode(mode: PlaybackMode): void {
    const wasPlaying = this._playing;
    if (wasPlaying) this.pause();
    this._mode = mode;
    this._onModeChange?.(mode);
    if (wasPlaying) this.play();
  }

  updateEventBlocks(blocks: EventBlock[]): void {
    this.eventBlocks = blocks;
    this.computeHeartbeatWindows();
  }

  play(): void {
    if (this._playing) return;
    this._playing = true;
    this.lastFrame = performance.now();

    if (this._mode === 'director' && this.autoFollow) {
      this.autoFollow.start();
    }

    const tick = (now: number) => {
      if (!this._playing || this.destroyed) return;
      const dt = now - this.lastFrame;
      this.lastFrame = now;

      const baseSpeed = this.timeMachine.currentSpeed;
      let effectiveSpeed = baseSpeed;

      if (this._mode === 'heartbeat') {
        effectiveSpeed = this.heartbeatSpeed(this.timeMachine.currentTimeMs) * baseSpeed;
      }

      const nextMs = this.timeMachine.currentTimeMs + dt * effectiveSpeed;
      this.timeMachine.setTimeExternal(nextMs);

      this.rafId = requestAnimationFrame(tick);
    };
    this.rafId = requestAnimationFrame(tick);
  }

  pause(): void {
    if (!this._playing) return;
    this._playing = false;
    if (this.rafId != null) { cancelAnimationFrame(this.rafId); this.rafId = null; }
    if (this._mode === 'director' && this.autoFollow) {
      this.autoFollow.stop();
    }
  }

  toggle(): void {
    if (this._playing) this.pause();
    else this.play();
  }

  destroy(): void {
    this.destroyed = true;
    this.pause();
  }

  private heartbeatSpeed(currentMs: number): number {
    // Find the density window containing currentMs
    for (const win of this.heartbeatWindows) {
      if (currentMs >= win.startMs && currentMs < win.endMs) {
        if (win.density === 0) return GlobePlayback.HB_MAX_SPEED;
        // High density = slow (linger), low density = fast
        const normalized = Math.min(win.density / 10, 1); // 10 events/h = max density
        const speed = GlobePlayback.HB_MAX_SPEED * (1 - normalized) + GlobePlayback.HB_MIN_SPEED * normalized;
        return speed;
      }
    }
    return 1; // default if no window found
  }

  private computeHeartbeatWindows(): void {
    if (this.eventBlocks.length === 0) {
      this.heartbeatWindows = [];
      return;
    }
    const minMs = Math.min(...this.eventBlocks.map(b => b.startMs));
    const maxMs = Math.max(...this.eventBlocks.map(b => b.endMs));
    const windows: HeartbeatWindow[] = [];
    for (let t = minMs; t < maxMs; t += GlobePlayback.HB_WINDOW_MS) {
      const windowEnd = t + GlobePlayback.HB_WINDOW_MS;
      const count = this.eventBlocks.filter(
        b => b.startMs < windowEnd && b.endMs > t,
      ).length;
      windows.push({
        startMs: t,
        endMs: windowEnd,
        density: count,
      });
    }
    this.heartbeatWindows = windows;
  }
}
```

```typescript
import {
  Cartesian3,
  Color,
  CustomDataSource,
  Entity,
  PolylineGraphics,
  CallbackProperty,
  type Viewer,
} from 'cesium';
import type { GlobeDataManager } from '../GlobeDataManager.js';

/** Trail configuration per layer type */
interface TrailConfig {
  layerName: string;
  windowMs: number;
  color: Color;
  width: number;
}

const TRAIL_CONFIGS: TrailConfig[] = [
  { layerName: 'flights', windowMs: 6 * 3_600_000, color: Color.fromCssColorString('#00c8ff'), width: 2 },
  { layerName: 'vessels', windowMs: 12 * 3_600_000, color: Color.fromCssColorString('#4488ff'), width: 1.5 },
  { layerName: 'darkVessels', windowMs: 12 * 3_600_000, color: Color.fromCssColorString('#ff4444'), width: 1.5 },
  { layerName: 'cyclones', windowMs: 7 * 86_400_000, color: Color.fromCssColorString('#ffa500'), width: 3 },
];

const MAX_TRAILS = 200;
const MAX_POINTS_PER_TRAIL = 50;

interface TrailEntry {
  entityId: string;
  layerName: string;
  /** Ring buffer: [lon0, lat0, timeMs0, lon1, lat1, timeMs1, ...] */
  buffer: Float64Array;
  head: number;
  count: number;
  cesiumEntity: Entity;
}

export class GlobeTrails {
  private viewer: Viewer;
  private dataManager: GlobeDataManager;
  private source: CustomDataSource;
  private trails: Map<string, TrailEntry> = new Map();
  private destroyed = false;
  private refreshTimer: ReturnType<typeof setInterval> | null = null;

  constructor(viewer: Viewer, dataManager: GlobeDataManager) {
    this.viewer = viewer;
    this.dataManager = dataManager;
    this.source = new CustomDataSource('4d-trails');
  }

  mount(): void {
    this.viewer.dataSources.add(this.source).catch(() => { /* intentional */ });
    this.refresh();
    // Refresh trail data every 5s
    this.refreshTimer = setInterval(() => { if (!this.destroyed) this.refresh(); }, 5000);
  }

  destroy(): void {
    this.destroyed = true;
    if (this.refreshTimer) { clearInterval(this.refreshTimer); this.refreshTimer = null; }
    this.viewer.dataSources.remove(this.source, true);
    this.trails.clear();
  }

  get activeTrailCount(): number { return this.trails.size; }

  private refresh(): void {
    const seenIds = new Set<string>();

    for (const config of TRAIL_CONFIGS) {
      const positions = this.dataManager.getEntityPositionHistory(config.layerName);
      for (const pos of positions) {
        if (this.trails.size >= MAX_TRAILS && !this.trails.has(pos.id)) continue;
        seenIds.add(pos.id);
        let trail = this.trails.get(pos.id);
        if (!trail) {
          trail = this.createTrail(pos.id, config);
          this.trails.set(pos.id, trail);
        }
        this.pushPoint(trail, pos.lon, pos.lat, pos.timeMs, config.windowMs);
      }
    }

    // Remove trails for entities no longer present
    for (const [id, trail] of this.trails) {
      if (!seenIds.has(id)) {
        this.source.entities.remove(trail.cesiumEntity);
        this.trails.delete(id);
      }
    }
  }

  private createTrail(entityId: string, config: TrailConfig): TrailEntry {
    const bufferSize = MAX_POINTS_PER_TRAIL * 3; // lon, lat, timeMs
    const buffer = new Float64Array(bufferSize);
    const trailColor = config.color.withAlpha(0.7);

    const cesiumEntity = this.source.entities.add(new Entity({
      polyline: new PolylineGraphics({
        positions: new CallbackProperty(() => {
          return this.getTrailPositions(entityId);
        }, false),
        width: config.width,
        material: new CallbackProperty(() => {
          return trailColor;
        }, false) as unknown as import('cesium').MaterialProperty,
        clampToGround: true,
      }),
    }));

    return { entityId, layerName: config.layerName, buffer, head: 0, count: 0, cesiumEntity };
  }

  private pushPoint(trail: TrailEntry, lon: number, lat: number, timeMs: number, windowMs: number): void {
    const idx = trail.head * 3;
    trail.buffer[idx] = lon;
    trail.buffer[idx + 1] = lat;
    trail.buffer[idx + 2] = timeMs;
    trail.head = (trail.head + 1) % MAX_POINTS_PER_TRAIL;
    if (trail.count < MAX_POINTS_PER_TRAIL) trail.count++;

    // Trim old points outside window
    const cutoff = Date.now() - windowMs;
    while (trail.count > 0) {
      const oldest = ((trail.head - trail.count + MAX_POINTS_PER_TRAIL) % MAX_POINTS_PER_TRAIL) * 3;
      if (trail.buffer[oldest + 2]! >= cutoff) break;
      trail.count--;
    }
  }

  private getTrailPositions(entityId: string): Cartesian3[] {
    const trail = this.trails.get(entityId);
    if (!trail || trail.count === 0) return [];
    const positions: Cartesian3[] = [];
    for (let i = 0; i < trail.count; i++) {
      const idx = ((trail.head - trail.count + i + MAX_POINTS_PER_TRAIL) % MAX_POINTS_PER_TRAIL) * 3;
      positions.push(Cartesian3.fromDegrees(trail.buffer[idx]!, trail.buffer[idx + 1]!));
    }
    return positions;
  }
}
```

## Task 8: GlobeDegradation — Reality Degradation Pipeline

**Files:**
- Create: `src/components/gods-eye/GlobeDegradation.ts`

Adjusts entity visual properties based on temporal confidence. Past events are crisp; future events progressively degrade.

- [ ] **Step 1: Create GlobeDegradation.ts**

```typescript
import {
  Color,
  type Viewer,
  JulianDate,
} from 'cesium';
import type { GlobeDataManager } from '../GlobeDataManager.js';

/** Per-layer maximum forecast window in ms */
const LAYER_FORECAST_MS: Record<string, number> = {
  conflicts: 7 * 86_400_000,
  airstrikes: 7 * 86_400_000,
  earthquakes: 30 * 86_400_000,
  gdacs: 2 * 86_400_000,
  cyclones: 5 * 86_400_000,
  fires: 2 * 86_400_000,
  flights: 0,
  vessels: 0,
  cyber: 0,
  gpsJamming: 0,
};

export interface DegradationState {
  confidence: number;
  opacity: number;
  isDashed: boolean;
  blur: number;
  jitter: number;
}

export class GlobeDegradation {
  private viewer: Viewer;
  private dataManager: GlobeDataManager;
  private cache: Map<string, DegradationState> = new Map();
  private destroyed = false;

  constructor(viewer: Viewer, dataManager: GlobeDataManager) {
    this.viewer = viewer;
    this.dataManager = dataManager;
  }

  /**
   * Recompute degradation for all visible entities relative to the given time.
   * Call this on time change (debounced 250ms by caller).
   */
  applyDegradation(currentMs: number): void {
    if (this.destroyed) return;
    this.cache.clear();

    const now = JulianDate.fromDate(new Date());
    const layerNames = Object.keys(LAYER_FORECAST_MS);

    for (const layerName of layerNames) {
      const maxForecast = LAYER_FORECAST_MS[layerName]!;
      if (maxForecast === 0) continue; // no forecast for this layer

      const entities = this.dataManager.getLayerEntitiesWithTimestamps(layerName);
      for (const e of entities) {
        const timeFromNow = e.timeMs - currentMs;

        if (timeFromNow <= 0) {
          // Past event — full confidence
          this.cache.set(e.id, { confidence: 1, opacity: 1, isDashed: false, blur: 0, jitter: 0 });
          continue;
        }

        const state = GlobeDegradation.computeState(timeFromNow, maxForecast);
        this.cache.set(e.id, state);

        // Apply opacity to the actual Cesium entity
        this.applyToEntity(layerName, e.id, state);
      }
    }
  }

  getState(entityId: string): DegradationState | undefined {
    return this.cache.get(entityId);
  }

  get forecastEntityCount(): number {
    let count = 0;
    for (const state of this.cache.values()) {
      if (state.confidence < 1) count++;
    }
    return count;
  }

  static computeState(timeFromNowMs: number, maxForecastMs: number): DegradationState {
    const confidence = Math.max(0.05, Math.min(1, 1 - timeFromNowMs / maxForecastMs));
    return {
      confidence,
      opacity: confidence,
      isDashed: confidence < 0.5,
      blur: confidence < 0.3 ? (1 - confidence) * 2 : 0,
      jitter: confidence < 0.2 ? (1 - confidence) * 4 : 0,
    };
  }

  private applyToEntity(layerName: string, entityId: string, state: DegradationState): void {
    // Find the entity in the data manager's layer source
    const sources = this.dataManager.getDataSources();
    const source = sources.get(layerName);
    if (!source) return;

    const entity = source.entities.getById(entityId);
    if (!entity) return;

    // Apply opacity via billboard/point alpha
    if (entity.billboard) {
      const currentColor = entity.billboard.color?.getValue(JulianDate.fromDate(new Date()));
      if (currentColor instanceof Color) {
        entity.billboard.color = currentColor.withAlpha(state.opacity) as unknown as import('cesium').Property;
      }
    }
    if (entity.point) {
      const currentColor = entity.point.color?.getValue(JulianDate.fromDate(new Date()));
      if (currentColor instanceof Color) {
        entity.point.color = currentColor.withAlpha(state.opacity) as unknown as import('cesium').Property;
      }
    }
    if (entity.label) {
      entity.label.show = state.confidence > 0.3 ? true as unknown as import('cesium').Property : false as unknown as import('cesium').Property;
    }
  }

  destroy(): void {
    this.destroyed = true;
    this.cache.clear();
  }
}
```

```typescript
 * Used by GlobePredictions and GlobeBranching for Tier 2 overlays.

/** A point on a predicted path */
export interface ForecastPoint {
  lat: number;
  lon: number;
  timeMs: number;
  confidence: number;
}

/** A probability cone (e.g., cyclone track forecast) */
export interface ForecastCone {
  entityId: string;
  type: 'cyclone' | 'flight' | 'fire' | 'vessel';
  probablePath: ForecastPoint[];
  conePolygon: { lat: number; lon: number }[]; // outer boundary
}

/** Aftershock probability model */
export interface AftershockForecast {
  entityId: string;
  epicenterLat: number;
  epicenterLon: number;
  mainshockMagnitude: number;
  rings: { radiusKm: number; probability: number; magnitudeThreshold: number }[];
}

/** Scenario branch for conflict escalation */
export interface ScenarioBranch {
  label: string;
  probability: number;
  outcome: 'escalation' | 'stalemate' | 'deescalation';
  path: ForecastPoint[];
  conditions: string;
}

export interface BranchingForecast {
  entityId: string;
  type: 'conflict' | 'cyclone';
  branches: ScenarioBranch[];
}

/**
 * Compute aftershock probability using simplified Reasenberg-Jones model.
 * P(M >= threshold | mainshock) over a 30-day window.
 */
export function computeAftershockForecast(
  entityId: string,
  lat: number,
  lon: number,
  magnitude: number,
): AftershockForecast {
  // Simplified ETAS: probability decreases with distance and magnitude threshold
  const rings: AftershockForecast['rings'] = [];
  const baseP = Math.min(0.95, Math.pow(10, magnitude - 5) * 0.1);

  for (const radiusKm of [25, 50, 100]) {
    const distanceFactor = 1 / (1 + radiusKm / 50);
    for (const magThreshold of [4, 5, 6]) {
      const magFactor = Math.pow(10, -(magThreshold - magnitude + 1));
      const prob = Math.min(0.95, Math.max(0.01, baseP * distanceFactor * magFactor));
      rings.push({ radiusKm, probability: prob, magnitudeThreshold: magThreshold });
    }
  }

  return { entityId, epicenterLat: lat, epicenterLon: lon, mainshockMagnitude: magnitude, rings };
}

/**
 * Compute conflict escalation branches using EMA trend direction.
 * Returns 3 scenario branches with probabilities.
 */
export function computeConflictBranches(
  entityId: string,
  lat: number,
  lon: number,
  emaTrend: number, // -1 to 1, negative = de-escalating, positive = escalating
): BranchingForecast {
  // Normalize trend to probabilities
  const escalateP = Math.max(0.05, Math.min(0.8, 0.3 + emaTrend * 0.4));
  const deescalateP = Math.max(0.05, Math.min(0.8, 0.3 - emaTrend * 0.4));
  const stalemateP = Math.max(0.05, 1 - escalateP - deescalateP);

  // Simple projected paths (bearing offsets from origin)
  const bearingStep = 0.005; // ~0.5km per step
  const makePathWithBearing = (bearing: number): ForecastPoint[] => {
    const points: ForecastPoint[] = [];
    for (let i = 0; i <= 7; i++) {
      const dayMs = i * 86_400_000;
      const confidence = Math.max(0.1, 1 - i / 7);
      points.push({
        lat: lat + Math.cos(bearing) * bearingStep * i,
        lon: lon + Math.sin(bearing) * bearingStep * i,
        timeMs: Date.now() + dayMs,
        confidence,
      });
    }
    return points;
  };

  return {
    entityId,
    type: 'conflict',
    branches: [
      {
        label: 'De-escalation',
        probability: deescalateP,
        outcome: 'deescalation',
        path: makePathWithBearing(Math.PI * 0.75),
        conditions: 'Ceasefire negotiations progress, external pressure increases',
      },
      {
        label: 'Stalemate',
        probability: stalemateP,
        outcome: 'stalemate',
        path: makePathWithBearing(0),
        conditions: 'Current tempo maintained, no significant territorial changes',
      },
      {
        label: 'Escalation',
        probability: escalateP,
        outcome: 'escalation',
        path: makePathWithBearing(-Math.PI * 0.5),
        conditions: 'New fronts open, external actors intervene, weapons escalation',
      },
    ],
  };
}

/**
 * Extrapolate a flight's future position along a great-circle heading.
 */
export function computeFlightCone(
  entityId: string,
  lat: number,
  lon: number,
  headingDeg: number,
  speedKnots: number,
): ForecastCone {
  const headingRad = (headingDeg * Math.PI) / 180;
  const speedKmH = speedKnots * 1.852;
  const points: ForecastPoint[] = [];
  const R = 6371; // Earth radius km

  for (let hours = 0; hours <= 2; hours += 0.25) {
    const distKm = speedKmH * hours;
    const dLat = (distKm / R) * Math.cos(headingRad) * (180 / Math.PI);
    const dLon = (distKm / R) * Math.sin(headingRad) * (180 / Math.PI) / Math.cos((lat * Math.PI) / 180);
    const confidence = Math.max(0.1, 1 - hours / 2);
    points.push({ lat: lat + dLat, lon: lon + dLon, timeMs: Date.now() + hours * 3_600_000, confidence });
  }

  // Cone polygon: expand laterally with time
  const conePolygon: { lat: number; lon: number }[] = [];
  const spreadRate = 0.002; // degrees per hour
  for (const p of points) {
    const hours = (p.timeMs - Date.now()) / 3_600_000;
    const spread = spreadRate * hours;
    const perpRad = headingRad + Math.PI / 2;
    conePolygon.push({ lat: p.lat + Math.cos(perpRad) * spread, lon: p.lon + Math.sin(perpRad) * spread });
  }
  for (let i = points.length - 1; i >= 0; i--) {
    const p = points[i]!;
    const hours = (p.timeMs - Date.now()) / 3_600_000;
    const spread = spreadRate * hours;
    const perpRad = headingRad - Math.PI / 2;
    conePolygon.push({ lat: p.lat + Math.cos(perpRad) * spread, lon: p.lon + Math.sin(perpRad) * spread });
  }

  return { entityId, type: 'flight', probablePath: points, conePolygon };
}
```

```typescript
    this.degradation = new GlobeDegradation(this.viewer, this.dataManager);

    // --- Tier 2: Deep Analysis (mounted but dormant) ---
    this.predictions = new GlobePredictions(this.viewer);
    this.predictions.mount();

    this.branching = new GlobeBranching(this.viewer);
    this.branching.mount();

    // Wire time changes to degradation + swimlane
    this.timeMachine.setOnTimeChange((ms, _live) => {
      this.degradation?.applyDegradation(ms);
      this.swimlane?.updateNow(ms);
    });

    // Refresh swimlane blocks periodically
    this.refreshBlocks();
    this.blockRefreshTimer = setInterval(() => this.refreshBlocks(), 15_000);

    // HUD update loop
    this.hudUpdateTimer = setInterval(() => this.updateHud(), 500);
    this.updateHud();

    // Hide existing simple scrubber
    const oldScrubber = this.container.querySelector('.godseye-time-machine') as HTMLElement | null;
    if (oldScrubber) oldScrubber.style.display = 'none';
  }

  deactivate(): void {
    if (!this._active) return;
    this._active = false;

    if (this.blockRefreshTimer) { clearInterval(this.blockRefreshTimer); this.blockRefreshTimer = null; }
    if (this.hudUpdateTimer) { clearInterval(this.hudUpdateTimer); this.hudUpdateTimer = null; }

    this.swimlane?.destroy(); this.swimlane = null;
    this.playback?.destroy(); this.playback = null;
    this.trails?.destroy(); this.trails = null;
    this.pillars?.destroy(); this.pillars = null;
    this.degradation?.destroy(); this.degradation = null;
    this.predictions?.destroy(); this.predictions = null;
    this.branching?.destroy(); this.branching = null;

    // Restore simple scrubber
    const oldScrubber = this.container.querySelector('.godseye-time-machine') as HTMLElement | null;
    if (oldScrubber) oldScrubber.style.display = '';

    // Clear time change listener
    this.timeMachine.setOnTimeChange(() => {});

    this.hud?.update4DState({
      active: false, playbackMode: 'documentary', speed: 1,
      playing: false, temporalOffsetMs: 0, trailCount: 0, forecastCount: 0,
    });
  }

  // --- Keyboard handlers (called by GodsEyeView) ---

  setPlaybackMode(mode: PlaybackMode): void {
    this.playback?.setMode(mode);
  }

  togglePlayback(): void {
    this.playback?.toggle();
  }

  toggleSwimlaneCollapse(): void {
    this.swimlane?.toggleCollapse();
  }

  zoomSwimlaneIn(): void {
    this.swimlane?.zoomIn();
  }

  zoomSwimlaneOut(): void {
    this.swimlane?.zoomOut();
  }

  snapToNow(): void {
    this.timeMachine.snapToNow();
  }

  dismissTier2(): void {
    this.predictions?.clear();
    this.branching?.clear();
  }

  // --- Internal ---

  private refreshBlocks(): void {
    const blocks = this.dataManager.getEventBlocks();
    this.swimlane?.updateBlocks(blocks);
    this.playback?.updateEventBlocks(blocks);
  }

  private handleEventClick(block: EventBlock): void {
    // Fly camera to event location
    if (block.lat != null && block.lon != null) {
      this.viewer.camera.flyTo({
        destination: Cartesian3.fromDegrees(block.lon, block.lat, 1_000_000),
        duration: 2,
      });
    }

    // Show Tier 2 analysis based on category
    this.dismissTier2();

    if (block.category === 'seismic' && block.lat != null && block.lon != null) {
      // Extract magnitude from description if available
      const magMatch = block.name.match(/M([\d.]+)/);
      const magnitude = magMatch ? parseFloat(magMatch[1]!) : 5;
      const forecast = computeAftershockForecast(block.id, block.lat, block.lon, magnitude);
      this.predictions?.showAftershock(forecast);
    } else if (block.category === 'conflicts' && block.lat != null && block.lon != null) {
      const branches = computeConflictBranches(block.id, block.lat, block.lon, 0);
      this.branching?.showBranches(branches);
    } else if (block.category === 'military' && block.lat != null && block.lon != null) {
      const cone = computeFlightCone(block.id, block.lat, block.lon, 0, 400);
      this.predictions?.showCone(cone);
    }
  }

  private updateHud(): void {
    this.hud?.update4DState({
      active: this._active,
      playbackMode: this.playback?.mode ?? 'documentary',
      speed: this.timeMachine.currentSpeed,
      playing: this.playback?.playing ?? false,
      temporalOffsetMs: this.timeMachine.currentTimeMs - Date.now(),
      trailCount: this.trails?.activeTrailCount ?? 0,
      forecastCount: this.degradation?.forecastEntityCount ?? 0,
    });
  }

  destroy(): void {
    this.deactivate();
  }
}
```
